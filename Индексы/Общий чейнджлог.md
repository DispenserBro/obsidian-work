# Добавлено

```dataview
TABLE rows.L.text AS "Логи"
FROM #logs
FLATTEN file.lists AS L
FLATTEN date(split(file.name, "_")[1]) AS dval
WHERE meta(L.section).subpath = "Added"
	AND L.text != ""
	AND dval > date("2025-10-01")
	AND dval < date("2025-10-31")
GROUP BY dateformat(dval, "dd MMM yyyy") AS "Дата"
```
